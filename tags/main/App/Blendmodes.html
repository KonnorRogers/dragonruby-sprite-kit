def self.src_over
  @src_over ||= Numeric.compose_blendmode(
    # RGB blending
    # src                           dst                       operator
    BLENDFACTOR_SRC_ALPHA, BLENDFACTOR_ONE_MINUS_SRC_ALPHA, BLENDOPERATION_ADD,
    # Alpha blending
    # src                           dst                       operator
    BLENDFACTOR_ONE,       BLENDFACTOR_ONE_MINUS_SRC_ALPHA, BLENDOPERATION_ADD
  )
